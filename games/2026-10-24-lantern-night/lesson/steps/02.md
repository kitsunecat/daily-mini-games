## このステップのゴール

ゲームの心臓部である **ゲームループ** を作り、広い夜の草原と、ランタンを持ったキツネを Canvas に描きます。タイトル画面の後ろでも草原とキツネが見えるようになります。

## 考え方: ワールドとカメラ

このゲームの舞台は一辺 3200 ピクセルの正方形の草原（**ワールド**）です。画面に映るのはその一部だけで、キツネを中心にした「窓」から見ています。この窓を **カメラ** と呼びます。

- キツネや敵の位置は、すべて **ワールド座標**（草原の左上が 0,0）で持ちます。
- 描くときにだけ「カメラの左上」を引いて画面の位置に直します。

こうしておくと、キツネが動けばカメラも一緒に動き、草原がスクロールして見えます。

## game.js を書き換える

追加する部分が多いので、`game.js` 全体を次の内容に置き換えます（`index.html` と `style.css` はそのままです）。

```js:game.js
// ランタン守りの夜 — 夜明けまで生き延びるサバイバー系アクション
(() => {
  'use strict';

  const WORLD = 3200;        // ワールドの一辺

  const canvas = document.getElementById('game');
  const ctx = canvas.getContext('2d');
  const $ = (id) => document.getElementById(id);

  const clamp = (v, a, b) => (v < a ? a : v > b ? b : v);
  const rand = (a, b) => a + Math.random() * (b - a);
  const TAU = Math.PI * 2;

  let W = 0, H = 0, DPR = 1, zoom = 1;
  function resize() {
    DPR = Math.min(window.devicePixelRatio || 1, 2);
    W = window.innerWidth;
    H = window.innerHeight;
    canvas.width = Math.round(W * DPR);
    canvas.height = Math.round(H * DPR);
    zoom = clamp(Math.min(W, H) / 620, 0.55, 1.25);
  }
  window.addEventListener('resize', resize);
  resize();

  // ---------- 状態 ----------
  let state = 'title'; // title / play
  let P;
  let t;

  function reset() {
    P = {
      x: WORLD / 2, y: WORLD / 2, r: 16,
      face: 1, walk: 0,
    };
    t = 0;
  }
  reset();

  // ---------- 更新 ----------
  function update(dt) {
    t += dt;
  }

  // ---------- 画面遷移 ----------
  function hideOverlays() {
    for (const id of ['title']) $(id).classList.add('hidden');
  }
  function startGame() {
    reset();
    hideOverlays();
    state = 'play';
  }

  $('startBtn').addEventListener('click', startGame);

  // ---------- 描画 ----------
  function hash(x, y) {
    let h = (x * 374761393 + y * 668265263) | 0;
    h = (h ^ (h >>> 13)) * 1274126177;
    return ((h ^ (h >>> 16)) >>> 0) / 4294967295;
  }

  function drawGround(camX, camY, vw, vh) {
    ctx.fillStyle = 'rgb(20,30,46)';
    ctx.fillRect(camX, camY, vw, vh);
    const T = 64;
    const x0 = Math.floor(camX / T), y0 = Math.floor(camY / T);
    const x1 = Math.ceil((camX + vw) / T), y1 = Math.ceil((camY + vh) / T);
    for (let ix = x0; ix <= x1; ix++) {
      for (let iy = y0; iy <= y1; iy++) {
        const h = hash(ix, iy);
        const px = ix * T + hash(iy, ix) * 40, py = iy * T + h * 40;
        if (h < 0.35) {
          ctx.strokeStyle = 'rgba(90,140,110,.35)';
          ctx.lineWidth = 2;
          ctx.beginPath();
          ctx.moveTo(px, py); ctx.lineTo(px - 4, py - 9);
          ctx.moveTo(px + 4, py); ctx.lineTo(px + 5, py - 11);
          ctx.moveTo(px + 8, py); ctx.lineTo(px + 12, py - 8);
          ctx.stroke();
        } else if (h < 0.42) {
          ctx.fillStyle = 'rgba(120,120,150,.35)';
          ctx.beginPath(); ctx.ellipse(px, py, 9, 6, 0, 0, TAU); ctx.fill();
        } else if (h < 0.46) {
          ctx.fillStyle = h < 0.44 ? 'rgba(255,190,220,.55)' : 'rgba(255,240,160,.55)';
          for (let k = 0; k < 5; k++) {
            const a = k * TAU / 5;
            ctx.beginPath(); ctx.arc(px + Math.cos(a) * 3.5, py + Math.sin(a) * 3.5, 2.4, 0, TAU); ctx.fill();
          }
        }
      }
    }
    // ワールドの外側（深い森）
    ctx.fillStyle = 'rgba(5,8,14,.85)';
    if (camX < 0) ctx.fillRect(camX, camY, -camX, vh);
    if (camY < 0) ctx.fillRect(camX, camY, vw, -camY);
    if (camX + vw > WORLD) ctx.fillRect(WORLD, camY, camX + vw - WORLD, vh);
    if (camY + vh > WORLD) ctx.fillRect(camX, WORLD, vw, camY + vh - WORLD);
  }

  function drawPlayer() {
    const x = P.x, y = P.y, f = P.face;
    const bob = P.moving ? Math.sin(P.walk) * 2 : 0;
    ctx.save();
    ctx.translate(x, y + bob);
    // 影
    ctx.fillStyle = 'rgba(0,0,0,.35)';
    ctx.beginPath(); ctx.ellipse(0, 16 - bob, 15, 5, 0, 0, TAU); ctx.fill();
    ctx.scale(f, 1);
    // しっぽ
    ctx.fillStyle = '#ff8a3d';
    ctx.beginPath(); ctx.ellipse(-16, 4, 11, 6, -0.6 + Math.sin(P.walk * 0.5) * 0.15, 0, TAU); ctx.fill();
    ctx.fillStyle = '#fff3e0';
    ctx.beginPath(); ctx.ellipse(-24, -1, 4, 3.5, -0.6, 0, TAU); ctx.fill();
    // 体
    ctx.fillStyle = '#ff9a4d';
    ctx.beginPath(); ctx.ellipse(0, 4, 12, 11, 0, 0, TAU); ctx.fill();
    ctx.fillStyle = '#fff3e0';
    ctx.beginPath(); ctx.ellipse(2, 7, 6, 6, 0, 0, TAU); ctx.fill();
    // 頭
    ctx.fillStyle = '#ffa55c';
    ctx.beginPath(); ctx.arc(3, -9, 10, 0, TAU); ctx.fill();
    ctx.beginPath(); ctx.moveTo(-5, -14); ctx.lineTo(-3, -25); ctx.lineTo(3, -17); ctx.fill();
    ctx.beginPath(); ctx.moveTo(5, -17); ctx.lineTo(11, -25); ctx.lineTo(12, -13); ctx.fill();
    ctx.fillStyle = '#fff3e0';
    ctx.beginPath(); ctx.ellipse(8, -6, 5, 4, 0, 0, TAU); ctx.fill();
    ctx.fillStyle = '#2b1a10';
    ctx.beginPath(); ctx.arc(7, -11, 1.8, 0, TAU); ctx.fill();
    ctx.beginPath(); ctx.arc(12.5, -7, 1.6, 0, TAU); ctx.fill();
    // ランタン
    ctx.strokeStyle = '#6b4a2b'; ctx.lineWidth = 2;
    ctx.beginPath(); ctx.moveTo(11, 2); ctx.lineTo(16, -4); ctx.stroke();
    ctx.fillStyle = '#ffd56b';
    ctx.shadowColor = '#ffb030'; ctx.shadowBlur = 18;
    ctx.beginPath(); ctx.roundRect ? ctx.roundRect(13, -3, 9, 11, 3) : ctx.rect(13, -3, 9, 11); ctx.fill();
    ctx.shadowBlur = 0;
    ctx.fillStyle = '#8a5a2e';
    ctx.fillRect(12, -4, 11, 2); ctx.fillRect(12, 7, 11, 2);
    ctx.restore();
  }

  function render() {
    ctx.setTransform(1, 0, 0, 1, 0, 0);
    const vw = W / zoom, vh = H / zoom;
    let camX = P.x - vw / 2, camY = P.y - vh / 2;
    const toWorld = () => ctx.setTransform(DPR * zoom, 0, 0, DPR * zoom, -camX * DPR * zoom, -camY * DPR * zoom);
    toWorld();
    drawGround(camX, camY, vw, vh);

    drawPlayer();
  }

  // ---------- ループ ----------
  let last = performance.now();
  function frame(now) {
    const dt = Math.min(0.05, (now - last) / 1000);
    last = now;
    if (state === 'play') update(dt);
    else if (state === 'title') { P.walk += dt * 4; t += dt * 0.2; }
    render();
    requestAnimationFrame(frame);
  }
  requestAnimationFrame(frame);
})();
```

上から順に、増えたところを見ていきましょう。

### 定数 WORLD と reset()

`WORLD = 3200` はワールドの一辺の長さです。`P` はプレイヤー（Player）のキツネで、位置 `x, y`、当たり判定の半径 `r`、向き `face`（1 で右向き、-1 で左向き）、歩きアニメーション用のカウンター `walk` を持たせます。`t` はゲーム開始からの経過秒数です。

`reset()` はこれらを初期値に戻す関数です。ゲームを始めるたびに `startGame()` から呼ぶので、何度遊び直しても最初の状態から始まります。今後、状態が増えたら `reset()` に初期化を足していきます。

### ゲームループ（requestAnimationFrame）

いちばん下の `frame` 関数がゲームループです。

1. `requestAnimationFrame(frame)` は「次に画面を描き替えるタイミングで `frame` を呼んで」とブラウザに頼む命令です。ふつうは1秒に60回ほど呼ばれます。
2. `frame` の最後でまた `requestAnimationFrame(frame)` を呼ぶので、ずっと繰り返されます。
3. 引数 `now` はミリ秒単位の時刻です。前回との差を秒に直したものが `dt`（delta time ＝ 前のフレームからの経過秒数）です。

**動きは必ず dt を掛けて計算します。** たとえば「1秒に 165 ピクセル進む」なら `165 * dt` ずつ進めます。こうすると 60Hz の画面でも 120Hz の画面でも同じ速さで動きます。`Math.min(0.05, ...)` は、タブを切り替えて戻ったときなどに `dt` が巨大になり、敵が一瞬でワープするのを防ぐための上限です。

`state` が `'play'` なら `update(dt)` でゲームを進め、`'title'` のときはしっぽを揺らす（`P.walk`）だけにしています。どちらの場合も `render()` で描画します。「更新（update）」と「描画（render）」を分けておくと、一時停止やタイトル画面でも絵だけは描き続けられます。

### render() と setTransform

`render()` ではまずカメラの左上 `camX, camY` を、キツネが画面の中央に来るように計算します。`vw, vh` は「ズームを考えたときに見えるワールドの幅と高さ」です。

`ctx.setTransform(a, b, c, d, e, f)` は、このあと描く図形の **拡大率と平行移動** をまとめて設定する命令です。`toWorld()` では

- 拡大率: `DPR * zoom`（高解像度対応とズーム）
- 平行移動: `-camX * DPR * zoom`（カメラの左上が画面の左上に来るようにずらす）

を指定しています。これ以降は **ワールド座標のまま** `ctx.arc(P.x, P.y, ...)` のように描けば、自動で画面上の正しい位置に描かれます。

### drawGround と hash

画面いっぱいを紺色で塗ったあと、64 ピクセルのマス目ごとに草・石・花を描きます。ここで `Math.random()` を使うと、フレームごとに模様が変わってチカチカしてしまいます。そこで `hash(x, y)` という「同じマス番号なら毎回同じ 0〜1 の値を返す」関数を使い、模様を固定しています。

- 見えている範囲のマスだけをループするので、ワールドが広くても重くなりません。
- 最後に、ワールドの外側（`camX < 0` など）を黒っぽく塗って「深い森」にします。端まで行くと見えます。

### drawPlayer

`ctx.save()` と `ctx.restore()` で囲むと、その間に変えた座標変換や色をあとで元に戻せます。

- `ctx.translate(x, y)` でキツネの位置を原点にしてから描くので、体の各パーツは「キツネの中心からの相対位置」で書けます。
- `ctx.scale(f, 1)` で、左向き（`f = -1`）のときは左右反転させます。右向きの絵を1つ描くだけで両方向に対応できます。
- `ellipse`（楕円）・`arc`（円）・`moveTo`/`lineTo`（三角の耳）を組み合わせてキツネを描いています。
- ランタンには `shadowBlur` を付けて、ぼんやり光って見えるようにしています。描いたらすぐ `shadowBlur = 0` に戻すのがコツです（戻さないと、そのあと描くものがすべてぼやけて重くなります）。
- `roundRect` は角丸の四角形です。古いブラウザには無いので、無ければ普通の `rect` を使うようにしています。
